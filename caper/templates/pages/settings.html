Usage examples: (full docs at docs.ampliconrepository.org)

# Set these once. A browser-style User-Agent header is required by the gateway.
TOKEN="<your-token>"
UA="Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Safari/537.36"

# List your projects
curl -s -A "$UA" -H "Authorization: Token $TOKEN" \
     https://ampliconrepository.org/api/v1/projects/

# Download a project archive (use -o, not -O; the URL ends in /download/)
curl -sS -L -A "$UA" -H "Authorization: Token $TOKEN" \
     -o "<id>.tar.gz" https://ampliconrepository.org/api/v1/projects/<id>/download/

# Get sample metadata
curl -s -A "$UA" -H "Authorization: Token $TOKEN" \
     https://ampliconrepository.org/api/v1/projects/<id>/samples/
